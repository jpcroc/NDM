  type atom_config ! type minimal des configurations atomiques. Tous les composants seront toujours alloué (im_glob seulement si PARA)
     integer::im=0  ! nombre d'atoems dans la cnfiguration
     real(double),allocatable:: xp(:,:) !xp(1:3,1:im) positions
     real(double),allocatable::::fp(:,:) ! forces
     integer,allocatable::ityp(:) ! types , initialisés à 0. Sert à détecter les atomes qui n'en sont pas (type=0= pas un vrai atome)
    integer,allocatable::ielat(:) ! indice de la cellule de l'atome
     logical :: ltabvois ! table des voisin ou pas
     integer,allocatable:: iwxmax(:) ! indice du dernier voisin de i (et pas nombre de voisin de i !)
     integer,allocatable:: indi(:) ! tableau de tous les voisins
     logical, allocatable:: lgul(:)    ! logical "General Utility Logical" peut servir de MASK 
     integer,allocatable::num_at_glob(:) ! indice global  toujours alloué et utilisé ; par défaut initiliser num_t_glob(i)=i
#ifdef PARA
     integer,allocatable::proc_at(:) ! tableau de taille im_glob total indiquant le numéro du proc qui gère l'atome
#endif     
   contains ! toutes ces procédures fonctionnent pour les 3 types
     procedure, pass::init=>init_atom_config ! initialisation/allocations
     procedure, pass::copy_atom=>copy_atom_config ! copy un atom d'une config à une autre
     procedure, pass::dealloc=>dealloc_atom_config ! dealloue les tableaux d'une config
     procedure, pass::copy_config ! copie toute une config dans une autre 
     procedure, pass::print ! print dégueu
     procedure, pass::pack ! efface les éléments de type 0 à la fin des tableaux la config
     procedure, pass::fab ! crée une config atcfextr extraite d'une autre à partir du MASK lgul
     procedure, pass::backto ! reverse la config crée par atextr dans atomconfig de départ (voir ci-dessous)
     procedure, pass::add2conf ! ajoute les éléments d'une configuration à la suite des éléments déja présents (<>copy_config)
     procedure, pass::extend ! étend la taille d'une configuration
     procedure, pass::send2proc=>s2p_atom ! envoi des éléments de la configuration vers un proc selon un communicateur
     procedure, pass::send2all=>s2a_atom ! envoi des éléments de la configuration vers tous les procs d'un communicateur
     procedure, pass::recv=>rcv_atom ! reçoit des éléments de la configuration depuis un proc selon un communicateur
     procedure, pass::zero=>zero_atom ! met une conf à zéro
     procedure, pass::vers_master=>vers_master_atom !(atcfloc,atcfcomp,div) envoi les configurations spatiales locales vers le master
     procedure, pass::master2loc=>master2loc_atom !(atcfcomp,atcfloc,div)  ! distribue la configuration complète vers lesprocs locauxde la parallélisation spatiale
     procedure, pass::switch_atom ! intervertit deux atomes de atcf
     procedure, pass::deftype ! initialise cible to the type of atsource, inluding the values of lax, lprteat, etc.
     procedure, pass::Eegal  ! pour atom_config_e initialise les variables logiques comme le modèle

  end type atom_config

  type, extends (atom_config):: atom_config_d ! type dynamique des configurations atomiques(+vp/+xpp). vp et xpp seront toujours allouées
     real(double),allocatable ::xpp(:,:) ! position pas précédent

  end type atom_config_d

call atsourcef%copy_atom(i,atcible,j,lextend=T ou F)
copie l'élément i de atsource dans l'élément j de atcible
atcible doit déja exister ET être initialisé
atsource et atcible n'ont pas à être de même type. Seules informations disponibles dans atsource ET existants dans atcible sont copiées

lextend=True ou False optionel, défaut = FALSE  !! (plus sur)
SI true : Si j est plus grand que la taile actelle de atcible(%im) on augmente la taille de at_cible
   permet de construite at_cible sans connaitre sa taille avant la construction
SI false : si j est plus grand que la taile actuelle de atcible  le code s'arrête
   permet devérifier que at_cible ne déborde pas

call atsource%copy_config(atcible,lrescl = T ou F)

copie toute la configuration source vers la configuration cible
atsource et atcible n'ont pas à être de même type. Seules informations disponibles dans atsource ET existants dans atcible sont copiées

atcible doit exister mais pas forcément être initialisé

lrescl : T ou F
       TRUE : si atcible n'a pas la taille de atsource : change automatiquement sa taille pour coller à atsource
       FALSE : si atcible est de taille plus petite que atsource : STOP. Sinon on copie atsource et on ne touche pas à ce qui dépasse dans atcible
