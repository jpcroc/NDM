  type atom_config ! type minimal des configurations atomiques. Tous les composants seront toujours alloué (im_glob seulement si PARA)
     integer::im=0  ! nombre d'atoems dans la cnfiguration
     real(double),allocatable:: xp(:,:) !xp(1:3,1:im) positions
     real(double),allocatable::::fp(:,:) ! forces
     integer,allocatable::ityp(:) ! types , initialisés à 0. Sert à détecter les atomes qui n'en sont pas (type=0= pas un vrai atome)
    integer,allocatable::ielat(:) ! indice de la cellule de l'atome
     logical :: ltabvois ! table des voisin ou pas
     integer,allocatable:: iwxmax(:) ! indice du dernier voisin de i (et pas nombre de voisin de i !)
     integer,allocatable:: indi(:) ! tableau de tous les voisins
     logical, allocatable:: lgul(:)    ! logical "General Utility Logical" peut servir de MASK 
     integer,allocatable::num_at_glob(:) ! indice global  toujours alloué et utilisé ; par défaut initiliser num_t_glob(i)=i
#ifdef PARA
     integer,allocatable::proc_at(:) ! tableau de taille im_glob total indiquant le numéro du proc qui gère l'atome
#endif     
   contains ! toutes ces procédures fonctionnent pour les 3 types
     procedure, pass::init=>init_atom_config ! initialisation/allocations
     procedure, pass::copy_atom=>copy_atom_config ! copy un atom d'une config à une autre
     procedure, pass::dealloc=>dealloc_atom_config ! dealloue les tableaux d'une config
     procedure, pass::copy_config ! copie toute une config dans une autre 
     procedure, pass::print ! print dégueu
     procedure, pass::pack ! efface les éléments de type 0 à la fin des tableaux la config
     procedure, pass::fab ! crée une extraite d'une autre à partir du MASK lgul
     procedure, pass::add2conf ! ajoute les éléments d'une configuration à la suite des éléments déja présents (<>copy_config)
     procedure, pass::extend ! étend la taille d'une configuration


  end type atom_config
